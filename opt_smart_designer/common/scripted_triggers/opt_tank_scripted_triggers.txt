# --- 陆军学说判定（HOI4 1.17+ 使用 has_doctrine；旧版请改为 has_tech）---

    has_doctrine = mobile_warfare
}

    has_doctrine = superior_firepower
}

    has_doctrine = grand_battleplan
}

    has_doctrine = mass_assault
}

# --- 坦克 MIO 可靠性补偿 ---
opt_has_tank_mio_reliability_bonus = {
    has_dlc = "Arms Against Tyranny"
    any_military_industrial_organization = {
        OR = {
            is_mio_trait_completed = generic_mio_trait_all_round_reliability
            is_mio_trait_completed = generic_mio_trait_simplified_turret_design
            is_mio_trait_completed = generic_mio_trait_standardized_components
            has_mio_size > 4
        }
    }
}

# 【陆军 - 机动战/装甲系】
opt_has_hired_mw_advisor = {
    any_character = {
        is_hired_as_advisor = yes
        OR = {
            has_trait = mobile_warfare_expert
            has_trait = blitzkrieg_theorist
            has_trait = army_armored_3
            has_trait = army_armored_2
        }
    }
}
opt_has_unhired_mw_advisor = {
    any_character = {
        is_hired_as_advisor = no
        OR = { has_advisor_role = theorist has_advisor_role = high_command has_advisor_role = army_chief }
        OR = {
            has_trait = mobile_warfare_expert
            has_trait = blitzkrieg_theorist
            has_trait = army_armored_3
            has_trait = army_armored_2
        }
    }
}
